import tensorflow as tf
import numpy as np
import matplotlib.pyplot as plt
(x_train, y_train), (x_test, y_test) = tf.keras.datasets.mnist.load_data()
#normalize and reshape 
x_train = x_train.astype('float32') / 255.0
x_test = x_test.astype('float32') / 255.0

x_train = x_train.reshape(-1, 784)
x_test = x_test.reshape(-1, 784)
#select normal data train on 0
x_train_normal = x_train[y_train == 0]

x_test_normal = x_test[y_test == 0]
x_test_anomaly = x_test[y_test != 0]
#encoder
encoder = tf.keras.Sequential([
    tf.keras.layers.Input(shape=(784,)),
    tf.keras.layers.Dense(128, activation='relu'),
    tf.keras.layers.Dense(32, activation='relu')
])
#decoder
decoder = tf.keras.Sequential([
    tf.keras.layers.Input(shape=(32,)),
    tf.keras.layers.Dense(128, activation='relu'),
    tf.keras.layers.Dense(784, activation='sigmoid')
])
autoencoder = tf.keras.Sequential([encoder, decoder])
#compile model
autoencoder.compile(
    optimizer='adam',
    loss='mse',
    metrics=['mae']
)
#Adam = Optimizer.
#MSE = Loss function
#MAE = Evaluation metric

import numpy as np
import tensorflow as tf
import matplotlib.pyplot as plt

from tensorflow import keras
from tensorflow.keras import layers
from collections import Counter
-
text = """
machine learning is very interesting
deep learning is a part of machine learning
artificial intelligence is changing the world
machine learning helps computers learn patterns
"""
words = text.lower().split()
print(words)
##  create vocabulary
vocab = sorted(set(words))
word_to_index = {
    word: i for i, word in enumerate(vocab)
}
index_to_word = {
    i: word for word, i in word_to_index.items()
}
vocab_size = len(vocab)

print("Vocabulary:", vocab)
print("Vocabulary Size:", vocab_size)
####set(words) removes duplicate words.
#sorted() sorts the words.
#word_to_index assigns a unique number to each word.
#index_to_word converts numbers back into words.
#vocab_size counts unique words.

#Generate CBOW training data
window_size = 2
data = []
for i in range(window_size, len(words) - window_size):

    context = (
        words[i-window_size:i] +
        words[i+1:i+window_size+1]
    )

    target = words[i]

    data.append((context, target))
print("Training Data:")
for context, target in data[:5]:
    print("Context:", context, "Target:", target)
##data.append() stores the input and output pair.

#Convert training data into numerical form
X = []
y = []
for context, target in data:
    context_indices = [
        word_to_index[word] for word in context
    ]

    target_index = word_to_index[target]

    X.append(context_indices)
    y.append(target_index)
X = np.array(X)
y = np.array(y)
print("Input Shape:", X.shape)
print("Output Shape:", y.shape)

#Define the CBOW model
model = keras.Sequential([

    layers.Embedding(
        input_dim=vocab_size,
        output_dim=50
    ),

    layers.Lambda(
        lambda x: tf.reduce_mean(x, axis=1)
    ),

    layers.Dense(
        vocab_size,
        activation='softmax'
    )
])
###Embedding	Converts word indexes into vectors
Lambda	Takes the average of context vectors
Dense + Softmax	Predicts the target word


model.compile(
    optimizer='adam',
    loss='sparse_categorical_crossentropy',
    metrics=['accuracy']
)


history = model.fit(
    X,
    y,
    epochs=100,
    batch_size=16,
    verbose=1
)

plt.plot(history.history['loss'])

plt.title("CBOW Model Training Loss")
plt.xlabel("Epochs")
plt.ylabel("Loss")

plt.show()

import tensorflow as tf
import numpy as np
import matplotlib.pyplot as plt

from tensorflow.keras import layers, models
base = tf.keras.applications.MobileNetV2(
    weights='imagenet',
    include_top=False,
    input_shape=(96, 96, 3)
)
#MobileNetV2 is our pre-trained CNN.
#ImageNet contains the original training images.
#include_top=False removes the original classifier.

Input image size is 96 × 96.

base.trainable = False

model = models.Sequential([
    layers.Input(shape=(96, 96, 3)),

    layers.Lambda(
        tf.keras.applications.mobilenet_v2.preprocess_input
    ),

    base,

    layers.GlobalAveragePooling2D(),

    layers.Dense(128, activation='relu'),

    layers.Dropout(0.3),

    layers.Dense(10, activation='softmax')
])
#preprocess_input: Prepares images for MobileNetV2.
#base: Extracts image features.
#GlobalAveragePooling2D: Reduces feature dimensions.
#Dense(128): Learns patterns.
#Dropout(0.3): Reduces overfitting.
#Dense(10): Predicts 10 categories.

(x_train, y_train), (x_test, y_test) = tf.keras.datasets.cifar10.load_data()


x_train = tf.image.resize(x_train, (96, 96))
x_test = tf.image.resize(x_test, (96, 96))

#Resizes images from 32 × 32 to 96 × 96 because MobileNetV2 is configured for 96 × 96 input images.

model.compile(
    optimizer='adam',
    loss='sparse_categorical_crossentropy',
    metrics=['accuracy']
)

#Adam: Optimizer.
#Sparse categorical crossentropy: Loss function.
#Accuracy: Measures correct predictions.

history = model.fit(
    x_train, y_train,
    epochs=3,
    batch_size=32,
    validation_data=(x_test, y_test)
)


#finetuning 
base.trainable = True

for layer in base.layers[:-20]:
    layer.trainable = False

for layer in base.layers:
    if isinstance(layer, layers.BatchNormalization):
        layer.trainable = False

##Unfreezes the base model.
Keeps the first layers frozen.
Makes the last 20 layers trainable.
Keeps Batch Normalization layers frozen for stable training.


model.compile(
    optimizer=tf.keras.optimizers.Adam(0.0001),
    loss='sparse_categorical_crossentropy',
    metrics=['accuracy']
)
##Recompiles the model with a smaller learning rate for fine-tuning.

history_fine = model.fit(
    x_train, y_train,
    epochs=2,
    batch_size=32,
    validation_data=(x_test, y_test)
)
##Trains the model again to adjust the selected pre-trained weights.


## Evaluate model final model using testing images
loss, accuracy = model.evaluate(x_test, y_test)

print("Test Accuracy:", accuracy)

graph##
plt.plot(history.history['accuracy'])
plt.plot(history_fine.history['accuracy'])

plt.title("Model Accuracy")
plt.xlabel("Epochs")
plt.ylabel("Accuracy")
plt.legend(["Initial Training", "Fine Tuning"])
plt.show()

# 1. Import libraries
import tensorflow as tf
from tensorflow.keras import layers, models

# 2. Load CIFAR-10 dataset
(x_train, y_train), (x_test, y_test) = tf.keras.datasets.cifar10.load_data()

# 3. Resize images
x_train = tf.image.resize(x_train, (96, 96))
x_test = tf.image.resize(x_test, (96, 96))

# 4. Load pre-trained MobileNetV2
base = tf.keras.applications.MobileNetV2(
    weights='imagenet',
    include_top=False,
    input_shape=(96, 96, 3)
)

# 5. Freeze base model
base.trainable = False

# 6. Add custom classifier
model = models.Sequential([
    layers.Input(shape=(96, 96, 3)),
    layers.Lambda(tf.keras.applications.mobilenet_v2.preprocess_input),
    base,
    layers.GlobalAveragePooling2D(),
    layers.Dense(128, activation='relu'),
    layers.Dense(10, activation='softmax')
])

# 7. Compile model
model.compile(
    optimizer='adam',
    loss='sparse_categorical_crossentropy',
    metrics=['accuracy']
)

# 8. Train model
model.fit(
    x_train, y_train,
    epochs=3,
    batch_size=32,
    validation_data=(x_test, y_test)
)

# 9. Fine-tuning: Unfreeze last 20 layers
base.trainable = True

for layer in base.layers[:-20]:
    layer.trainable = False

# 10. Compile again with smaller learning rate
model.compile(
    optimizer=tf.keras.optimizers.Adam(0.0001),
    loss='sparse_categorical_crossentropy',
    metrics=['accuracy']
)

# 11. Train again
model.fit(
    x_train, y_train,
    epochs=2,
    batch_size=32,
    validation_data=(x_test, y_test)
)

# 12. Evaluate model
loss, accuracy = model.evaluate(x_test, y_test)

print("Test Loss:", loss)
print("Test Accuracy:", accuracy)
